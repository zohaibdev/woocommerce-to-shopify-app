# Jagah:  public_html/app/backend/.htaccess
#
# Kyun: backend/ web root ke andar hai, is liye .env, vendor/, app/, storage/
# browser se khul saktay hain. Ye file un sab ko block karti hai.

<IfModule mod_authz_core.c>
    Require all denied
</IfModule>

<IfModule !mod_authz_core.c>
    Order deny,allow
    Deny from all
</IfModule>

Options -Indexes
